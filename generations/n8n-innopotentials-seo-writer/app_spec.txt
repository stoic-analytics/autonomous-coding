<project_specification>
  <project_name>n8n-innopotentials-seo-writer</project_name>

  <overview>
    An n8n workflow automation that transforms external URLs and PDF documents into SEO-optimized WordPress blog posts for the DACH region (Germany, Austria, Switzerland). The system uses a Google Sheets interface for client input and approval, with AI-powered content generation and image creation running locally.
  </overview>

  <technology_stack>
    <automation>
      <platform>n8n</platform>
      <hosting>Self-hosted Docker</hosting>
    </automation>
    <ai>
      <llm>Ollama (local)</llm>
      <image_generation>Flux Schnell (local, Apple Silicon M2 Max)</image_generation>
    </ai>
    <integrations>
      <input>Google Sheets API + Google Apps Script</input>
      <output>WordPress REST API + Application Password</output>
      <notifications>Email (SMTP)</notifications>
    </integrations>
    <wordpress>
      <theme>Spacious</theme>
      <post_status>Draft</post_status>
    </wordpress>
  </technology_stack>

  <prerequisites>
    <environment_setup>
      - Docker and Docker Compose installed
      - n8n container running and accessible
      - Ollama installed with suitable model (e.g., llama3, mistral)
      - Flux Schnell set up via ComfyUI or diffusers on M2 Max
      - Google Cloud project with Sheets API enabled
      - WordPress site with Application Password created
      - SMTP credentials for email notifications
    </environment_setup>
  </prerequisites>

  <feature_count>49</feature_count>

  <workflow_stages>
    <stage number="1" name="Input">
      <description>Client enters source data in Google Sheets</description>
      <components>
        - Google Sheet with structured columns
        - URL field for source website (required)
        - PDF links field for supplementary documents (optional)
        - Language selector (German/English)
        - Status tracking column
        - Approve button (Google Apps Script)
      </components>
    </stage>

    <stage number="2" name="Parse">
      <description>Extract content from provided sources</description>
      <components>
        - Web scraper node for URL content extraction
        - PDF parser for document text extraction
        - Content cleaning and normalization
        - Error handling for unreachable URLs or corrupt PDFs
      </components>
    </stage>

    <stage number="3" name="Generate Content">
      <description>AI creates SEO-optimized blog post</description>
      <components>
        - Ollama HTTP API integration
        - DACH region keyword analysis prompt
        - SEO-optimized blog post generation
        - Category and tag suggestion
        - Meta description generation
        - Language-appropriate output (German or English)
      </components>
    </stage>

    <stage number="4" name="Generate Image">
      <description>Create featured image for blog post</description>
      <components>
        - Flux Schnell via ComfyUI API or diffusers
        - Prompt generation based on blog content
        - Image optimization for web (compression, resizing)
        - Multiple aspect ratio support
      </components>
    </stage>

    <stage number="5" name="Review">
      <description>Client reviews and approves content</description>
      <components>
        - Generated content written back to Google Sheet
        - Status updated to "Ready for Review"
        - Client proofreads and edits if needed
        - Approve button triggers webhook to n8n
      </components>
    </stage>

    <stage number="6" name="Publish">
      <description>Push approved content to WordPress</description>
      <components>
        - WordPress REST API integration
        - Create post as draft
        - Upload and attach featured image
        - Assign categories and tags
        - Set SEO metadata
        - Update sheet status to "Published"
      </components>
    </stage>
  </workflow_stages>

  <google_sheets_structure>
    <columns>
      <column name="ID" type="auto">Auto-incrementing row identifier</column>
      <column name="URL" type="text">Source website URL to parse (required)</column>
      <column name="PDF_Links" type="text">Comma-separated PDF URLs (optional)</column>
      <column name="Language" type="dropdown">German | English</column>
      <column name="Status" type="text">Pending | Processing | Ready for Review | Approved | Published | Failed</column>
      <column name="Error_Message" type="text">Error details if failed</column>
      <column name="Generated_Title" type="text">AI-generated blog post title</column>
      <column name="Generated_Content" type="text">AI-generated blog post body</column>
      <column name="Generated_Keywords" type="text">Extracted DACH keywords</column>
      <column name="Generated_Categories" type="text">Suggested WordPress categories</column>
      <column name="Generated_Tags" type="text">Suggested WordPress tags</column>
      <column name="Image_URL" type="text">Generated image URL or path</column>
      <column name="Approve" type="button">Google Apps Script button to trigger publish</column>
      <column name="WordPress_URL" type="text">Published draft URL</column>
      <column name="Created_At" type="timestamp">Row creation timestamp</column>
      <column name="Updated_At" type="timestamp">Last update timestamp</column>
    </columns>
  </google_sheets_structure>

  <n8n_workflow_nodes>
    <trigger_nodes>
      <node name="Google Sheets Trigger">Webhook triggered by new row or Apps Script</node>
      <node name="Approval Webhook">Receives approval button click from Apps Script</node>
    </trigger_nodes>

    <processing_nodes>
      <node name="HTTP Request - Scrape URL">Fetch and parse website content</node>
      <node name="HTTP Request - Download PDF">Download PDF documents</node>
      <node name="Code - PDF Parser">Extract text from PDF using pdf-parse or similar</node>
      <node name="Code - Content Cleaner">Normalize and clean extracted text</node>
      <node name="HTTP Request - Ollama">Send prompt to local Ollama API</node>
      <node name="HTTP Request - Flux/ComfyUI">Generate image via local Flux</node>
      <node name="Code - Image Optimizer">Compress and resize image for web</node>
      <node name="Google Sheets - Update">Write generated content back to sheet</node>
      <node name="HTTP Request - WordPress">Create draft post via REST API</node>
      <node name="HTTP Request - WordPress Media">Upload featured image</node>
      <node name="Send Email">Notification on failure</node>
    </processing_nodes>

    <utility_nodes>
      <node name="IF - Has PDF">Branch if PDF links provided</node>
      <node name="IF - Success/Failure">Error handling branches</node>
      <node name="Set - Variables">Configure dynamic values</node>
      <node name="Merge">Combine parallel execution paths</node>
    </utility_nodes>
  </n8n_workflow_nodes>

  <api_integrations>
    <google_sheets>
      <auth>OAuth2 or Service Account</auth>
      <scopes>spreadsheets, drive.readonly</scopes>
      <operations>read, update, append</operations>
    </google_sheets>

    <google_apps_script>
      <purpose>Approval button webhook trigger</purpose>

    </google_apps_script>

    </flux_comfyui>

    <wordpress>
      <endpoint>https://{site}/wp-json/wp/v2/</endpoint>

      <endpoints>

        - POST /media (upload image)
        - GET /categories (list categories)
        - GET /tags (list tags)
      </endpoints>
    </wordpress>

    <email>
      <protocol>SMTP</protocol>
      <purpose>Failure notifications</purpose>
    </email>
  </api_integrations>

  <llm_prompts>
    <keyword_extraction>
      Analyze the following content and extract 5-10 SEO keywords relevant to the DACH region (Germany, Austria, Switzerland). Consider local search trends, German language variations, and regional business terminology.

      Content: {parsed_content}
      Language: {language}

      Output format: comma-separated keywords
    </keyword_extraction>

    <blog_post_generation>
      You are an expert SEO content writer for the DACH region. Write a comprehensive, SEO-optimized blog post based on the following source material.

      Source Content: {parsed_content}
      Target Keywords: {keywords}
      Language: {language}

      Requirements:
      - Write in {language}
      - Include target keywords naturally (2-3% density)
      - Structure with H2 and H3 headings
      - Include introduction and conclusion
      - 800-1500 words
      - Engaging, professional tone
      - Optimized meta description (150-160 characters)

      Output JSON format:
      {
        "title": "...",
        "meta_description": "...",
        "content": "...",
        "suggested_categories": ["..."],
        "suggested_tags": ["..."]
      }
    </blog_post_generation>

    <image_prompt_generation>
      Based on this blog post, generate a Flux image prompt for a professional featured image:

      Title: {title}
      Keywords: {keywords}

      Create a prompt for a modern, professional blog header image. Avoid text in the image. Style: clean, corporate, relevant to the topic.
    </image_prompt_generation>
  </llm_prompts>

  <error_handling>
    <strategy>Mark failed in sheet + email notification</strategy>
    <error_types>
      <error type="url_unreachable">
        <action>Set status to "Failed", record error message</action>
        <notification>Yes</notification>
      </error>
      <error type="pdf_parse_failure">
        <action>Set status to "Failed", record error message</action>
        <notification>Yes</notification>
      </error>
      <error type="llm_timeout">
        <action>Set status to "Failed", record error message</action>
        <notification>Yes</notification>
      </error>
      <error type="image_generation_failure">
        <action>Continue without image, note in sheet</action>
        <notification>No</notification>
      </error>
      <error type="wordpress_api_error">
        <action>Set status to "Failed", record error message</action>
        <notification>Yes</notification>
      </error>
    </error_types>
  </error_handling>

  <processing_mode>

  </processing_mode>
